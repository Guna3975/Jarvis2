<script setup>
import { ref } from 'vue'

defineProps({
  message: String
})

const emit = defineEmits(['add-leave'])

const leaveId = ref('')
const empId = ref('')
const type = ref('')
const status = ref('')

function addLeave() {
  emit('add-leave', {
    leaveId: leaveId.value,
    empId: empId.value,
    type: type.value,
    status: status.value
  })

  leaveId.value = ''
  empId.value = ''
  type.value = ''
  status.value = ''
}
</script>

<template>
  <div>
    <h2>Add Leave</h2>

    <form @submit.prevent="addLeave">
      <input v-model="leaveId" placeholder="Leave ID" />
      <input v-model="empId" placeholder="Employee ID" />
      <input v-model="type" placeholder="Leave Type" />
      <input v-model="status" placeholder="Status" />

      <button type="submit">Add Leave</button>
    </form>

    <p>{{ message }}</p>
  </div>
</template>



<script setup>
import { ref } from 'vue'

const emit = defineEmits([
  'update-balance',
  'delete-leave'
])

const empId = ref('')
const newBal = ref(0)
const leaveId = ref('')

function updateBalance() {
  emit('update-balance', {
    empId: empId.value,
    leaveBal: newBal.value
  })
}

function deleteLeave() {
  emit('delete-leave', {
    leaveId: leaveId.value
  })
}
</script>

<template>
  <div>
    <h2>Update Leave Balance</h2>

    <form @submit.prevent="updateBalance">
      <input v-model="empId" placeholder="Employee ID" />
      <input
        v-model.number="newBal"
        type="number"
        placeholder="New Leave Balance"
      />
      <button type="submit">Update Balance</button>
    </form>

    <h2>Delete Leave</h2>

    <form @submit.prevent="deleteLeave">
      <input v-model="leaveId" placeholder="Leave ID" />
      <button type="submit">Delete Leave</button>
    </form>
  </div>
</template>



<script setup>
import { ref, onMounted } from 'vue'
import LeaveForm from './components/LeaveForm.vue'
import LeaveActions from './components/LeaveActions.vue'

const url = 'http://localhost:8080/api/leave'

const leaves = ref([])
const message = ref('')

async function getLeaves() {
  const response = await fetch(`${url}/sick`)
  leaves.value = await response.json()
}

async function addLeave(leave) {
  const response = await fetch(`${url}/add`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json'
    },
    body: JSON.stringify(leave)
  })

  if (response.ok) {
    message.value = 'Leave added successfully'
    await getLeaves()
  } else {
    message.value = 'Failed to add leave'
  }
}

async function updateBalance(employee) {
  const response = await fetch(
    `${url}/update/${employee.empId}`,
    {
      method: 'PUT',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        leaveBal: employee.leaveBal
      })
    }
  )

  message.value = response.ok
    ? 'Leave balance updated successfully'
    : 'Failed to update leave balance'
}

async function deleteLeave(leave) {
  const response = await fetch(
    `${url}/delete/${leave.leaveId}`,
    {
      method: 'DELETE'
    }
  )

  if (response.ok) {
    message.value = 'Leave deleted successfully'
    await getLeaves()
  } else {
    message.value = 'Failed to delete leave'
  }
}

onMounted(getLeaves)
</script>

<template>
  <h1>Leave Management System</h1>

  <LeaveForm
    :message="message"
    @add-leave="addLeave"
  />

  <hr />

  <LeaveActions
    @update-balance="updateBalance"
    @delete-leave="deleteLeave"
  />

  <p>{{ message }}</p>

  <hr />

  <h2>Leave Records</h2>

  <button @click="getLeaves">Refresh Leaves</button>

  <div class="leave-grid">
    <div
      v-for="leave in leaves"

      :class="leave.status?.toLowerCase()"
    >

      <p><strong>Employee ID:</strong> {{ leave.empId }}</p>
      <p><strong>Type:</strong> {{ leave.type }}</p>
      <p><strong>Status:</strong> {{ leave.status }}</p>
    </div>
  </div>
</template>
